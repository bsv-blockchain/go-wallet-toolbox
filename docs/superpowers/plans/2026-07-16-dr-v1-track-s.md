```go
// testSchemaName builds a valid, unique postgres schema name (<=63 chars) from the test name.
func testSchemaName(t testing.TB) string {
	sanitized := strings.ToLower(regexp.MustCompile(`[^a-z0-9_]+`).ReplaceAllString(strings.ToLower(t.Name()), "_"))
	suffix := make([]byte, 4)
	_, _ = rand.Read(suffix)
	name := fmt.Sprintf("t_%s_%x", sanitized, suffix)
	if len(name) > 63 {
		name = name[:50] + name[len(name)-13:]
	}
	return name
}
```

```go
package testabilities

import (
	"testing"

	"github.com/stretchr/testify/assert"

// AssertStorageInvariants checks cross-table money-safety invariants that must
// hold after ANY storage scenario. Extend as waves land (Decision Record v1, W0-3).
func AssertStorageInvariants(t testing.TB, db *gorm.DB) {
	t.Helper()

	var n int64

	// 1. An output is never simultaneously spendable and spent.
	assert.NoError(t, db.Raw(`SELECT count(*) FROM bsv_outputs WHERE spendable = true AND spent_by IS NOT NULL`).Scan(&n).Error)
	assert.Zerof(t, n, "%d outputs are spendable AND spent_by-claimed", n)

	// 2. No completed user transaction without a merkle proof on its KnownTx.
	assert.NoError(t, db.Raw(`
		SELECT count(*) FROM bsv_transactions tx
		JOIN bsv_known_txs ktx ON ktx.tx_id = tx.tx_id
		WHERE tx.status = 'completed' AND ktx.merkle_path IS NULL`).Scan(&n).Error)
	assert.Zerof(t, n, "%d completed transactions lack a merkle proof", n)

	// 3. A UserUTXO row (funder-spendable coin) must project a live output.
	assert.NoError(t, db.Raw(`
		SELECT count(*) FROM bsv_user_utxos uu
		JOIN bsv_outputs o ON o.id = uu.output_id
		WHERE o.spendable = false OR o.spent_by IS NOT NULL`).Scan(&n).Error)
	assert.Zerof(t, n, "%d user_utxos rows point at dead outputs", n)
}
```

**Interfaces:**
- Consumes: `gocron.Locker` / `gocron.Lock` interfaces (`github.com/go-co-op/gocron/v2`): `Lock(ctx context.Context, key string) (gocron.Lock, error)`; `Lock.Unlock(ctx) error`. gocron skips the run and reschedules when `Lock` errors (verified: executor.go:449-473 in the module cache).
- Produces:
  - `func NewLeaseLocker(db *gorm.DB, owner string, logger *slog.Logger) (*LeaseLocker, error)` — AutoMigrates its table.
  - `func (l *LeaseLocker) SetLeaseTTL(jobName string, ttl time.Duration)` — called from `Daemon.Start` once task intervals are known (`TTL = max(2*interval, 5*time.Minute)`).
  - `NewDaemonWithGORMLocker(ctx, logger, storage, db, opts...)` — signature unchanged; internally uses `LeaseLocker` instead of `gocron-gorm-lock`'s 1ms wall-clock-bucket identifier (`monitor.go:73`, vacuous: pods never collide on a key).

Verified library context: `gocron-gorm-lock` excludes only via `UNIQUE(job_name, job_identifier)` where the identifier is acquisition-time `time.Now().Truncate(1ms)`; RUNNING rows are immortal (sweep deletes FINISHED only); slot-truncated keys are ALSO vacuous because pods' duration-job schedules aren't aligned. Lease CAS is the correct portable shape.

Model + semantics:

```go
// monitorJobLock is one row per job name; ownership = holding an unexpired lease.
type monitorJobLock struct {
	JobName    string    `gorm:"primaryKey"`
	Owner      string    `gorm:"not null"`
	LeaseUntil time.Time `gorm:"not null;index"`
	UpdatedAt  time.Time
}

func (l *LeaseLocker) Lock(ctx context.Context, key string) (gocron.Lock, error) {
	now := time.Now().UTC()
	ttl := l.ttlFor(key) // default 10m when unset
	// ensure row exists (portable across sqlite/postgres/mysql)
	_ = l.db.WithContext(ctx).Clauses(clause.OnConflict{DoNothing: true}).
		Create(&monitorJobLock{JobName: key, Owner: l.owner, LeaseUntil: now.Add(ttl)})
	// claim: CAS on expired lease (also covers the row we may have just created —
	// re-claim it to distinguish our insert from a concurrent one)
	res := l.db.WithContext(ctx).Model(&monitorJobLock{}).
		Where("job_name = ? AND (owner = ? OR lease_until < ?)", key, l.owner, now).
		Updates(map[string]any{"owner": l.owner, "lease_until": now.Add(ttl)})
	if res.Error != nil {
		return nil, fmt.Errorf("lease claim failed for %s: %w", key, res.Error)
	}
	if res.RowsAffected == 0 {
		l.logger.WarnContext(ctx, "monitor job skipped: lease held by another instance",
			slog.String("job", key))
		return nil, fmt.Errorf("lease for %s held by another instance", key)
	}
	return &leaseHandle{locker: l, key: key}, nil
}

// leaseHandle.Unlock releases early: sets lease_until = now WHERE owner = me.
```

Caveat to encode in a comment: `owner = ?` in the claim means the same instance re-claims its own lease (correct: gocron won't overlap runs of one job in one process by default; the lease guards *across* processes). Crash reclaim: another pod claims once `lease_until < now`. Clock skew tolerance = TTL slack; document "pods must be NTP-sane within minutes".

TTL wiring: `Daemon.Start(ctx, enabledTasks)` already receives `map[defs.MonitorTask]defs.TaskConfig` — before scheduling, when the locker is a `*LeaseLocker`, call `SetLeaseTTL(taskName, max(2*cfg.Interval(), 5*time.Minute))`. Keep a `leaseLocker *LeaseLocker` field on Daemon (nil when constructed via the plain `NewDaemon` without locker).
